cmake_minimum_required(VERSION 3.16)
project(CoursePlanner LANGUAGES CXX)
find_package(SQLite3 REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(planner src/Repository.cpp src/Database.cpp src/Tree.cpp src/Service.cpp src/Menu.cpp)
target_include_directories(planner PUBLIC include)
if(TARGET SQLite3::SQLite3)
  target_link_libraries(planner PUBLIC SQLite3::SQLite3)
else()
  target_link_libraries(planner PUBLIC SQLite::SQLite3)
endif()
if(MSVC)
  target_compile_options(planner PRIVATE /W4)
else()
  target_compile_options(planner PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(course_planner src/main.cpp)
target_link_libraries(course_planner PRIVATE planner)
enable_testing()
add_executable(planner_tests tests/tests.cpp)
target_link_libraries(planner_tests PRIVATE planner)
add_test(NAME planner_tests COMMAND planner_tests)
add_executable(tree_benchmark benchmarks/tree_benchmark.cpp)
target_link_libraries(tree_benchmark PRIVATE planner)
